a raw SQL query (text)
WITH ordered AS ( SELECT token_id, ending_price, ROW_NUMBER() OVER (PARTITION BY token_id ORDER BY timestamp ASC) AS rn_first, ROW_NUMBER() OVER (PARTITION BY token_id ORDER BY timestamp DESC) AS rn_last FROM transactions WHERE lower(status)='completed' AND timestamp >= (SELECT MAX(timestamp) FROM transactions) - INTERVAL '7 days') SELECT f.token_id, f.ending_price AS start_price, l.ending_price AS end_price, (l.ending_price - f.ending_price)/NULLIF(f.ending_price,0) AS price_growth_rate FROM ordered f JOIN ordered l ON f.token_id=l.token_id WHERE f.rn_first=1 AND l.rn_last=1 ORDER BY price_growth_rate DESC
